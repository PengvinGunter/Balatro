create table role(
role_id int primary key identity(1,1) not null,
role_name nvarchar(50) not null
)
create table payment_met(
id_method int primary key identity(1,1) not null,
method_name nvarchar(50) not null,
)
create table users(
user_id int primary key identity(1,1) ,
user_name nvarchar(50) not null,
user_secondname nvarchar(50) not null,
user_phone nvarchar(20) not null,
user_email nvarchar(70) unique not null,
password nvarchar(50) not null,
role_id int not null,
foreign key (role_id) references role(role_id),
date_registration datetime not null,
)
create table envir_class(
id_class int primary key identity(1,1),
class_name nvarchar(50)not null,
)

create table fuel(
id_fuel int primary key identity(1,1),
grade_fuel nvarchar (40)not null,
id_class int not null,
foreign key (id_class) references envir_class(id_class),
cost_per_litr money not null,
remains_fuel float not null,
)

create table gas_station(
id_station int primary key identity(1,1),
station_name nvarchar(100) not null,
station_city nvarchar(50) not null,
station_street nvarchar (50) not null,
station_housr nvarchar(50) not null,
station_quantity int not null,
station_worktime nvarchar(50) not null
)

create table sales(
id_sale int primary key identity(1,1),
date_sale datetime not null,
id_gas_station int not null,
foreign key (id_gas_station) references gas_station(id_station),
user_id int not null,
foreign key (user_id) references users(user_id),
id_fuel int not null,
foreign key (id_fuel) references fuel(id_fuel),
volume_fuel float not null,
cost money not null,
id_method int not null,
foreign key (id_method) references payment_met(id_method),
)
